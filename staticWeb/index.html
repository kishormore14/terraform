DECLARATIVE INFRASTRUCTURE

Readable. Reproducible. Production-Ready.

resource "random_string" "bucket_name" {
  length  = 8
  special = false
  upper   = true
}

resource "aws_s3_bucket" "demobucket" {
  bucket        = "my-kishor-bucket-${lower(random_string.bucket_name.result)}-123"
  force_destroy = true

  tags = {
    Name = "my-kishor-bucket-staticweb"
  }
}

resource "aws_s3_object" "demo_file" {
  bucket = aws_s3_bucket.demobucket.id
  key    = "/kishor/index.html"
  source = "./index.html"
}
